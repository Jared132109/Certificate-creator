    <script>
        function addStudent() {
            var roll = document.getElementById("roll").value;
            var name = document.getElementById("name").value;
            var studentClass = document.getElementById("class").value;

            var sub1 = document.getElementById("sub1").value;
            var marks1 = parseInt(document.getElementById("marks1").value);
            var sub2 = document.getElementById("sub2").value;
            var marks2 = parseInt(document.getElementById("marks2").value);
            var sub3 = document.getElementById("sub3").value;
            var marks3 = parseInt(document.getElementById("marks3").value);
            var sub4 = document.getElementById("sub4").value;
            var marks4 = parseInt(document.getElementById("marks4").value);
            var sub5 = document.getElementById("sub5").value;
            var marks5 = parseInt(document.getElementById("marks5").value);
            var sub6 = document.getElementById("sub6").value;
            var marks6 = parseInt(document.getElementById("marks6").value);

            // Calculate total
            var total = marks1 + marks2 + marks3 + marks4 + marks5 + marks6;

            var table = document.getElementById("studentTable");
            var row = table.insertRow();
            row.insertCell(0).innerHTML = roll;
            row.insertCell(1).innerHTML = name;
            row.insertCell(2).innerHTML = studentClass;
            row.insertCell(3).innerHTML = sub1;
            row.insertCell(4).innerHTML = marks1;
            row.insertCell(5).innerHTML = sub2;
            row.insertCell(6).innerHTML = marks2;
            row.insertCell(7).innerHTML = sub3;
            row.insertCell(8).innerHTML = marks3;
            row.insertCell(9).innerHTML = sub4;
            row.insertCell(10).innerHTML = marks4;
            row.insertCell(11).innerHTML = sub5;
            row.insertCell(12).innerHTML = marks5;
            row.insertCell(13).innerHTML = sub6;
            row.insertCell(14).innerHTML = marks6;
            row.insertCell(15).innerHTML = total;

            // Clear form after adding
            document.getElementById("studentForm").reset();
        }
    </script>
